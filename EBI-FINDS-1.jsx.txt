function ImgFallback({ src, alt, className }) {
  const [broken, setBroken] = useState(false);
  if (broken) {
    return (
      <div
        className={className}
        style={{
          background: `linear-gradient(135deg, ${COLORS.beige}, ${COLORS.rose})`,
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
        }}
      >
        <Sparkles size={28} color={COLORS.ink} strokeWidth={1.3} />
      </div>
    );
  }
  return (
    <img
      src={src}
      alt={alt}
      loading="lazy"
      onError={() => setBroken(true)}
      className={className}
    />
  );
}
